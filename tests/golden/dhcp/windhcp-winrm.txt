initialize
--- winrm
{"auth": ["svc", "pw"], "endpoint": "http://dhcp01:5985/wsman", "transport": "ntlm"}
--- winrm-script
$ErrorActionPreference = 'Stop'
$p = ConvertFrom-Json '{"ScopeId": "10.0.0.0", "IPAddress": "10.0.0.10", "ClientId": "aa-bb-cc-dd-ee-ff", "Name": "web01", "ComputerName": "", "Options": [{"Id": 3, "Value": ["10.0.0.1"]}, {"Id": 6, "Value": ["10.0.0.53", "10.0.0.54"]}, {"Id": 15, "Value": ["example.com"]}, {"Id": 1, "Value": ["255.255.255.0"]}, {"Id": 28, "Value": ["10.0.0.255"]}, {"Id": 67, "Value": ["undionly.kpxe"]}]}'
$common = @{}
if ($p.ComputerName) { $common['ComputerName'] = $p.ComputerName }
Add-DhcpServerv4Reservation -ScopeId $p.ScopeId -IPAddress $p.IPAddress -ClientId $p.ClientId -Name $p.Name @common
foreach ($o in $p.Options) {
  Set-DhcpServerv4OptionValue -ReservedIP $p.IPAddress -OptionId $o.Id -Value $o.Value @common
}
Write-Verbose extra
ConvertTo-Json @{ ok = $true }

--- step
complete
--- winrm
{"auth": ["svc", "pw"], "endpoint": "http://dhcp01:5985/wsman", "transport": "ntlm"}
--- winrm-script
$ErrorActionPreference = 'Stop'
$p = ConvertFrom-Json '{"ScopeId": "10.0.0.0", "ClientId": "aa-bb-cc-dd-ee-ff", "ComputerName": ""}'
$common = @{}
if ($p.ComputerName) { $common['ComputerName'] = $p.ComputerName }
$existing = Get-DhcpServerv4Reservation -ScopeId $p.ScopeId @common -ErrorAction SilentlyContinue | Where-Object { $_.ClientId -eq $p.ClientId }
if ($existing) {
  Remove-DhcpServerv4Reservation -ScopeId $p.ScopeId -ClientId $p.ClientId @common
}
ConvertTo-Json @{ ok = $true }
