# netboot: create the DHCP policy 'ipxe' once, as an admin
$scope = '10.0.0.0'
if (-not (Get-DhcpServerv4Class -Name 'iPXE' -Type User -ComputerName 'dhcp01' -ErrorAction SilentlyContinue)) { Add-DhcpServerv4Class -Name 'iPXE' -Type User -Data 'iPXE' -ComputerName 'dhcp01' }
Add-DhcpServerv4Policy -Name 'ipxe' -ScopeId $scope -Condition AND -UserClass EQ,'iPXE' -ComputerName 'dhcp01'
Set-DhcpServerv4OptionValue -PolicyName 'ipxe' -ScopeId $scope -OptionId 67 -Value 'boot.ipxe' -ComputerName 'dhcp01'
